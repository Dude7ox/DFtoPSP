cmake_minimum_required(VERSION 3.16)
project(DFtoPSP CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dfpsp_core src/core/Vfs.cpp src/core/Bsa.cpp src/core/Texture.cpp src/core/Arch3d.cpp src/core/Rdb.cpp src/core/Maps.cpp src/core/Scene.cpp src/core/Dungeon.cpp src/core/Font.cpp src/core/Ui.cpp src/core/Automap.cpp src/core/Cull.cpp src/core/Midi.cpp src/core/Synth.cpp src/core/Sfx.cpp src/core/Audio.cpp src/core/SoftwareRenderer.cpp src/core/Enemy.cpp src/core/EnemyData.cpp src/core/SaveGame.cpp)
target_include_directories(dfpsp_core PUBLIC src)
add_executable(dfpsp_host src/platform/host/main.cpp)
target_link_libraries(dfpsp_host dfpsp_core)
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core dfpsp_core)
enable_testing()
add_test(NAME core COMMAND test_core ${CMAKE_SOURCE_DIR}/data)
